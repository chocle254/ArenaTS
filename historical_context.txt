          {step === 'info' && (
            <>
              <Button variant="ghost" onClick={() => onOpenChange(false)} className="sm:flex-1">Not Now</Button>
              <Button onClick={() => setStep('id_upload')} className="sm:flex-1 font-bold">Start Verification</Button>
            </>

          {step === 'id_upload' && (
            <>

            </>

          {step === 'face_verification' && (
            <>
              <Button variant="ghost" onClick={() => setStep('id_upload')} className="sm:flex-1">Back</Button>
              <div className="flex flex-col sm:flex-row gap-2 sm:flex-[2]">

      </div>
    </Dialog>
